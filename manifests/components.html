function FigmaEditorAtomHydrator({
  nodes,
  edges = [],
  children,
}: {
  nodes: RoadmapNode[];
  edges?: Edge[];
  children: React.ReactNode;
}) {
  useHydrateAtoms([
    [nodesAtom, nodes],
    [edgesAtom, edges],
  ]);

  return children;
}
